export\_data() {

log "开始导出数据到: $OUTPUT\_FILE"

local start\_time=$(date +%s)

\# 修复：移除数组字段（manufacturer），避免错误

EXPORT\_QUERY='

SELECT

order\_id,

customer\_id,

customer\_full\_name,

order\_date,

total\_quantity,

taxful\_total\_price,

'

local end\_time=$(date +%s)

local duration=$((end\_time - start\_time))

\# 检查导出是否成功

local line\_count=$(wc -l < "$OUTPUT\_FILE")

export\_with\_curl() {

}

EOF

info "通过 curl 执行 SQL 查询..."

\-H "Content-Type: application/json" \\

\-u "elastic:elastic" \\

\-d @/tmp/sql\_query.json > "$OUTPUT\_FILE" 2>> "$LOG\_FILE"; then

local duration=$((end\_time - start\_time))

local line\_count=$(wc -l < "$OUTPUT\_FILE")

return 1

fi

else

return 1

fi

}

main() {

logger.error(f"合并分片文件失败 {slice\_file}: {e}")

except Exception as e:

logger.error(f"合并文件失败: {e}")

raise

def parallel\_export(self, index\_name, query\_body, output\_file, num\_slices=8):

"""并行导出主函数"""

start\_time = datetime.now()

try:

\# 获取分片配置

slices, total\_docs = self.get\_shard\_ranges(index\_name, query\_body, num\_slices)

if total\_docs == 0:

logger.warning("没有找到文档可导出")

return

logger.info(f"开始并行导出，使用 {num\_slices} 个分片")

slice\_files = \[\]

total\_processed = 0

with ThreadPoolExecutor(max\_workers=self.max\_workers) as executor:

future\_to\_slice = {

executor.submit(

self.export\_slice,

index\_name, slice\_query, i, output\_file

): i for i, slice\_query in enumerate(slices)

}

for future in as\_completed(future\_to\_slice):

slice\_id = future\_to\_slice\[future\]

try:

slice\_file, slice\_count = future.result()

if slice\_file:

slice\_files.append(slice\_file)

total\_processed += slice\_count

logger.info(f"分片 {slice\_id} 处理完成: {slice\_count} 条记录")

except Exception as e:

logger.error(f"分片 {slice\_id} 执行异常: {e}")

\# 合并文件

if slice\_files:

self.merge\_slice\_files(slice\_files, output\_file)

end\_time = datetime.now()

duration = (end\_time - start\_time).total\_seconds()

logger.info(f"✅ 并行导出完成: {total\_processed}/{total\_docs} 条记录，耗时 {duration:.2f} 秒")

logger.info(f"📁 输出文件: {output\_file}")

\# 显示文件信息

if os.path.exists(output\_file):

file\_size = os.path.getsize(output\_file)

logger.info(f"📄 文件大小: {file\_size / 1024 / 1024:.2f} MB")

else:

logger.error("没有分片成功处理")

except Exception as e:

logger.error(f"❌ 并行导出失败: {str(e)}")

raise

\# 使用示例

脚本内容如下：

#!/bin/bash

set -e

echo "开始部署 Logstash 8.17.9..."

\# 创建目录

mkdir -p /home/ubuntu/logstash/{config,pipeline,patterns,data,logs}

\# 创建配置文件

cat > /home/ubuntu/logstash/config/logstash.yml << 'EOF'

http.host: "0.0.0.0"

path.data: /usr/share/logstash/data

path.logs: /usr/share/logstash/logs

\# 监控配置

xpack.monitoring.enabled: true

xpack.monitoring.elasticsearch.hosts: \["http://localhost:9200"\]

xpack.monitoring.elasticsearch.username: elastic

xpack.monitoring.elasticsearch.password: elastic

xpack.monitoring.elasticsearch.ssl.verification\_mode: none

\# 配置重载

config.reload.automatic: true

config.reload.interval: 10s

EOF

cat > /home/ubuntu/logstash/config/pipelines.yml << 'EOF'

\- pipeline.id: main

path.config: "/usr/share/logstash/pipeline/\*.conf"

pipeline.workers: 2

pipeline.batch.size: 125

EOF

cat > /home/ubuntu/logstash/pipeline/main.conf << 'EOF'

input {

port => 5000

codec => json\_lines

}

port => 8081

codec => json

}

}

filter {

}

mutate {

remove\_field => \[ "headers" \]

}

}

output {

hosts => \["http://localhost:9200"\]

user => "elastic"

password => "elastic"

index => "logstash-%{+YYYY.MM.dd}"

ssl => false

ssl\_certificate\_verification => false

}

stdout {

}

\# 创建空的自定义模式文件（可选）

\# 拉取镜像

sudo docker pull elastic/logstash:8.17.9

\# 停止并删除旧容器（如果存在）

sudo docker stop logstash-8.17.9 || true

sudo docker rm logstash-8.17.9 || true

\# 启动新容器

sudo docker run -d \\

\--name logstash-8.17.9 \\

\--restart unless-stopped \\

\--memory=2g \\

\--cpus=2 \\

\-p 5044:5044 \\

\-p 5000:5000 \\

\-p 5001:5001 \\

\-p 8081:8081 \\

\-p 9600:9600 \\

\-v /home/ubuntu/logstash/config:/usr/share/logstash/config \\

\-v /home/ubuntu/logstash/patterns:/usr/share/logstash/patterns \\

\-v /home/ubuntu/logstash/data:/usr/share/logstash/data \\

\-v /home/ubuntu/logstash/logs:/usr/share/logstash/logs \\

\-e LS\_JAVA\_OPTS="-Xms512m -Xmx512m" \\

elastic/logstash:8.17.9

echo "等待容器启动..."

sleep 30

\# 验证部署

if curl -s http://localhost:9600/ > /dev/null; then

echo "✅ Logstash 部署成功!"

echo "📊 监控地址: http://localhost:9600/"

echo "🔌 TCP 输入端口: 5000"

echo "🌐 HTTP 输入端口: 8081"

else

fi
