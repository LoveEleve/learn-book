## 1. 时间函数

### 1.1 `gettimeofday` vs `clock_gettime`

```c
#include <sys/time.h>

struct timeval tv;
gettimeofday(&tv, NULL);              // wall-clock——系统管理员可调
printf("Wall clock: %ld.%06ld\n", tv.tv_sec, tv.tv_usec);

#include <time.h>

struct timespec ts;
clock_gettime(CLOCK_MONOTONIC, &ts);  // 单调递增——不受系统时间跳变影响
printf("Monotonic: %ld.%09ld\n", ts.tv_sec, ts.tv_nsec);
```

**为什么 `CLOCK_MONOTONIC` 必须替代 `gettimeofday`**：系统管理员可以用 `date -s` 调整系统时间——`gettimeofday` 的值会跳变。如果一个服务用 `gettimeofday` 算超时，系统时间前跳 1 小时 → 超时变成负值 → 死循环。Redis/Nginx 全部用 `CLOCK_MONOTONIC` 做超时。

| 时钟类型 | 含义 | 适用场景 |
|---|---|---|
| `CLOCK_REALTIME` | 墙钟——可被管理员调整（NTP 步进/时区变更） | 面向人类的时间戳 |
| `CLOCK_MONOTONIC` | **单调递增**——不受系统时间调整影响 | 超时/延时/性能测量（**所有服务端代码必须用**） |
| `CLOCK_MONOTONIC_RAW` | 底层硬���计数器——不受 NTP 微调影响 | 极精确的性能基准测试 |

### 1.2 `nanosleep`——精确延时

```c
struct timespec ts = { .tv_sec = 0, .tv_nsec = 500000000 };  // 500ms
nanosleep(&ts, NULL);
// 被信号中断时，剩余时间写入第二个参数——可以继续休眠
```

`usleep` 已过时（POSIX 2008 弃用）。`nanosleep` 是标准替代。

### 1.3 POSIX 定时器——取代 `alarm`

```c
timer_t timerid;
struct sigevent sev = { .sigev_notify = SIGEV_SIGNAL, .sigev_signo = SIGUSR1 };
timer_create(CLOCK_MONOTONIC, &sev, &timerid);

struct itimerspec its = { .it_value = { .tv_sec = 1 } };  // 首次 1 秒后
timer_settime(timerid, 0, &its, NULL);
// 1 秒后向当前进程发送 SIGUSR1
// epoll 集成：用 timerfd_create 替代——timerfd 是 epoll loop 的定时器首选
```

## 2. /proc 文件系统

### 2.1 进程信息：`/proc/PID/`

```c
// 从 C 代码中读取进程资源（不用 popen("ps")）
FILE *fp = fopen("/proc/self/status", "r");
char line[256];
while (fgets(line, sizeof(line), fp)) {
    if (strncmp(line, "VmRSS:", 6) == 0)  printf("%s", line);  // 常驻内存
    if (strncmp(line, "Threads:", 8) == 0) printf("%s", line);  // 线程数
    if (strncmp(line, "voluntary_ctxt_switches:", 24) == 0) printf("%s", line);
}
fclose(fp);
```

**高频读取的 /proc/PID 文件**：

| 文件 | 内容 | 生产用途 |
|---|---|---|
| `status` | 内存（VmRSS/VmSize）、线程数、信号掩码 | 监控常驻内存 |
| `stat` | CPU 时间（utime+stime）、优先级、状态 | 计算 CPU 利用率 |
| `fd/` | 打开的文件描述符列表 | S4Ch01 §4.6 fd 泄漏诊断 |
| `maps` | 地址空间映射 | S3Ch03 §4.4 共享库布局 |
| `limits` | 资源限制（NOFILE/NPROC 等） | 调优前检查 |
| `io` | 磁盘 I/O 统计 | 确认读写量 |

### 2.2 系统级信息：`/proc/`

```c
// /proc/meminfo——监控脚本的标配
FILE *fp = fopen("/proc/meminfo", "r");
// MemTotal: / MemAvailable: / SwapTotal:

// /proc/cpuinfo——core 数 + 型号
fopen("/proc/cpuinfo", "r");
```

## 3. 文件锁：`flock` vs `fcntl`

### 3.1 `flock`——BSD 文件锁

```c
#include <sys/file.h>

int fd = open("/tmp/lockfile", O_RDWR | O_CREAT, 0644);

flock(fd, LOCK_EX);       // 排他锁——阻塞
// 临界区
flock(fd, LOCK_UN);       // 释放

// 非阻塞版本
if (flock(fd, LOCK_EX | LOCK_NB) == -1) {
    // 锁已被持有——退避
}
```

**特点**：只能锁整个文件，基于内核 `struct file`——同一个 fd 多次 `dup`/`fork` 共享同一把锁。

### 3.2 `fcntl`——POSIX 记录锁

```c
#include <fcntl.h>

struct flock lk = {
    .l_type   = F_WRLCK,   // 排他写锁
    .l_whence = SEEK_SET,
    .l_start  = 0,
    .l_len    = 100        // 仅锁文件的前 100 字节
};
fcntl(fd, F_SETLKW, &lk);  // 阻塞等锁
```

**特点**：可以锁文件的一部分（字节范围），基于进程 PID——`fork` 后父子不共享同一把锁。

### 3.3 SQLite 用文件锁做并发控制

SQLite 通过 `flock` 实现多读单写：

```c
flock(fd, LOCK_SH);    // 多个读者——共享锁可以同时持有
flock(fd, LOCK_EX);    // 单个写者——排他锁
// WAL 模式切换：sqlite3WalDefaultHook → 文件锁状态决定读写并发
```

## 4. 目录操作

```c
#include <dirent.h>

DIR *d = opendir("/tmp");
struct dirent *entry;
while ((entry = readdir(d)) != NULL) {
    if (entry->d_type == DT_REG)           // 普通文件
        printf("File: %s\n", entry->d_name);
    else if (entry->d_type == DT_DIR)       // 目录
        printf("Dir:  %s\n", entry->d_name);
}
closedir(d);

// 递归遍历：nftw("/path", callback, 10, 0);
// 获取当前目录：getcwd(buf, sizeof(buf));
// 切换目录：chdir("/tmp");
```

**`scandir`——过滤 + 排序的组合**：

```c
#include <dirent.h>

int filter_txt(const struct dirent *e) {
    return strstr(e->d_name, ".txt") != NULL;  // 返回 1 保留，0 丢弃
}

struct dirent **namelist;
int n = scandir("/tmp", &namelist, filter_txt, alphasort);
for (int i = 0; i < n; i++) {
    printf("%s\n", namelist[i]->d_name);
    free(namelist[i]);
}
free(namelist);
```

## 5. 用户/组/权限

```c
#include <unistd.h>

uid_t uid  = getuid();      // 实际用户 ID
uid_t euid = geteuid();     // 有效用户 ID（setuid 程序用）
gid_t gid  = getgid();

// setuid 程序：从 root 降权（Nginx master 用）
if (geteuid() == 0) {
    setgid(65534);     // nobody
    setuid(65534);
    // 现在这个进程不再是 root——安全性提升
}

// 检查能力（Linux capabilities）：
#include <sys/capability.h>
cap_t caps = cap_get_proc();
cap_flag_value_t val;
cap_get_flag(caps, CAP_NET_BIND_SERVICE, CAP_PERMITTED, &val);
printf("Can bind to port <1024: %s\n", val ? "yes" : "no");
```

**`umask`——设置文件创建权限掩码**：

```c
mode_t old = umask(0027);   // 所有者 rwx，组 r-x，其他人无
int fd = open("new.txt", O_CREAT | O_WRONLY, 0666);  // 实际权限 = 0666 & ~0027 = 0640
```

## 6. 动态加载：`dlopen`/`dlsym`

```c
#include <dlfcn.h>

void *handle = dlopen("libmylib.so", RTLD_NOW);  // 立即解析所有符号
if (!handle) { fprintf(stderr, "%s\n", dlerror()); exit(1); }

typedef int (*add_fn)(int, int);
add_fn add = dlsym(handle, "my_add");             // 按函数名查找
if (!add) { fprintf(stderr, "%s\n", dlerror()); exit(1); }

printf("100+200=%d\n", add(100, 200));

dlclose(handle);
```

**`RTLD_NOW` vs `RTLD_LAZY`**：
- `RTLD_NOW`：立即解析所有符号——找不到立即失败
- `RTLD_LAZY`：按需解析——找不到只在首次调用时失败

**`RTLD_GLOBAL` 标志**：导出的符号对其他动态库可见——Nginx 用它暴露 `ngx_cycle` 给模块。

## 7. UDP + `getaddrinfo`

```c
#include <sys/socket.h>
#include <netdb.h>

// getaddrinfo——协议无关的地址解析
struct addrinfo hints = {
    .ai_family = AF_UNSPEC,      // 同时支持 IPv4 和 IPv6
    .ai_socktype = SOCK_DGRAM,   // UDP
};
struct addrinfo *res;
getaddrinfo("time.google.com", "123", &hints, &res);

// 用第一个可用的地址创建 socket
int fd = socket(res->ai_family, res->ai_socktype, 0);

// UDP send/recv
char buf[48] = {0};
sendto(fd, buf, 48, 0, res->ai_addr, res->ai_addrlen);  // 发送 NTP 请求

struct sockaddr_storage src;
socklen_t addrlen = sizeof(src);
recvfrom(fd, buf, sizeof(buf), 0, (struct sockaddr *)&src, &addrlen);

freeaddrinfo(res);
close(fd);
```

**UDP vs TCP**：UDP 是数据报——每个 `sendto`/`recvfrom` 是一个独立的边界。TCP 是字节流——消息边界由应用层协议维护（S4Ch04 §5）。UDP 不需 accept/connect（虽然 `connect` 可以绑定默认对端）。

## 8. 临时文件与环境变量

```c
// mkstemp——原子创建临时文件（避免竞态条件）
char tmpl[] = "/tmp/myapp-XXXXXX";
int fd = mkstemp(tmpl);  // "XXXXXX" 被替换为随机字符，返回 fd
write(fd, "temp data", 9);

// unlink immediately——文件在 close 前仍可用，但文件系统上不可见
unlink(tmpl);
// fd 仍然有效，直到 close(fd)

// 环境变量
printf("HOME=%s\n", getenv("HOME"));

setenv("MY_VAR", "hello", 1);   // 1=覆盖已有
unsetenv("MY_VAR");

// 遍历所有环境变量
extern char **environ;
for (char **env = environ; *env; env++) {
    printf("%s\n", *env);
}
```

**`mkstemp` 为什么安全**：以前的 `tmpnam` 在"生成名字"和"创建文件"之间有竞态条件——另一个进程可以抢先创建同名文件。`mkstemp` 原子地完成这两步。

## 9. 总结

| 主题 | 核心生产知识 |
|---|---|
| `CLOCK_MONOTONIC` | 所有超时/延时必须用——`gettimeofday` 会跳变 |
| `/proc/PID/status` | VmRSS/Threads/上下文切换——监控脚本标配 |
| `flock` | BSD 文件锁——同一 fd 共享，SQLite 并发控制基础 |
| `fcntl(F_SETLK)` | POSIX 记录锁——字节范围、进程级 |
| `opendir/readdir/scandir` | 目录遍历——Nginx 配置目录、cron 遍历任务文件 |
| `geteuid/setuid` | root 降权——Nginx master→worker 模式 |
| `dlopen/dlsym` | 动态加载——Nginx 模块、PostgreSQL 扩展 |
| `getaddrinfo` | 协议无关——同时支持 IPv4/IPv6 |
| `mkstemp` | 原子创建临时文件——防竞态条件 |
